127[[nodiscard]] inline std::vector<Complex> spherical_bessel_j(int max_order, Complex z) {
128 if (max_order < 0) throw InvalidArgument("spherical_bessel_j: max_order must be non-negative");
129 std::vector<Complex> j(static_cast<std::size_t>(max_order) + 1, Complex{0.0, 0.0});
130 if (std::abs(z) < 1e-300) {
131 j[0] = Complex{1.0, 0.0};
132 return j;
133 }
134 // start the recurrence where j_n is negligible relative to j_N; values grow while descending
135 const int start = max_order + 20 + static_cast<int>(std::ceil(std::abs(z)));
136 Complex next{0.0, 0.0}; // f_{n+1}
137 Complex current{1e-300, 0.0}; // f_n, arbitrary scale
138 for (int n = start; n > 0; --n) {
139 const Complex previous = static_cast<Real>(2 * n + 1) / z * current - next; // f_{n-1}
140 next = current;
141 current = previous;
142 if (n - 1 <= max_order) j[static_cast<std::size_t>(n - 1)] = current;
143 if (std::abs(current) > 1e200) { // rescale against overflow
144 constexpr Real kScale = 1e-200;
145 current *= kScale;
146 next *= kScale;
147 for (auto& v : j) v *= kScale;
148 }
149 }
150 const Complex scale = std::sin(z) / z / j[0];
151 for (auto& v : j) v *= scale;
152 return j;
153}

162[[nodiscard]] inline std::vector<Real> bessel_i_scaled(int max_order, Real x) {
163 if (max_order < 0) throw InvalidArgument("bessel_i_scaled: max_order must be non-negative");
164 if (!(x >= 0)) throw InvalidArgument("bessel_i_scaled: x must be non-negative");
165 std::vector<Real> out(static_cast<std::size_t>(max_order) + 1, 0.0);
166 if (x < 1e-150) { // I_n(x) = (x/2)^n / n! + …: below 1e-150 for n ≥ 1
167 out[0] = 1.0;
168 return out;
169 }
170 const int start = max_order + 32 + static_cast<int>(std::ceil(std::sqrt(80.0 * x)));
171 Real next = 0.0; // I_{n+1}
172 Real current = 1e-300; // I_n, arbitrary scale
173 Real sum = 0.0; // 2 Σ_{k ≥ n} I_k
174 for (int n = start; n > 0; --n) {
175 sum += 2.0 * current;
176 const Real previous = 2.0 * n / x * current + next; // I_{n-1}
177 next = current;
178 current = previous;
179 if (n - 1 <= max_order) out[static_cast<std::size_t>(n - 1)] = current;
180 if (current > 1e200) { // rescale against overflow
181 constexpr Real kScale = 1e-200;
182 current *= kScale;
183 next *= kScale;
184 sum *= kScale;
185 for (auto& v : out) v *= kScale;
186 }
187 }
188 sum += current; // + I_0
189 for (auto& v : out) v /= sum;
190 return out;
191}
