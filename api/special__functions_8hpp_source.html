70 if (max_order < 0) throw InvalidArgument("spherical_bessel_j: max_order must be non-negative");
71 std::vector<Complex> j(static_cast<std::size_t>(max_order) + 1, Complex{0.0, 0.0});
72 if (std::abs(z) < 1e-300) {
73 j[0] = Complex{1.0, 0.0};
74 return j;
75 }
76 // start the recurrence where j_n is negligible relative to j_N; values grow while descending
77 const int start = max_order + 20 + static_cast<int>(std::ceil(std::abs(z)));
78 Complex next{0.0, 0.0}; // f_{n+1}
79 Complex current{1e-300, 0.0}; // f_n, arbitrary scale
80 for (int n = start; n > 0; --n) {
81 const Complex previous = static_cast<Real>(2 * n + 1) / z * current - next; // f_{n-1}
82 next = current;
83 current = previous;
84 if (n - 1 <= max_order) j[static_cast<std::size_t>(n - 1)] = current;

87 current *= kScale;
88 next *= kScale;
89 for (auto& v : j) v *= kScale;
