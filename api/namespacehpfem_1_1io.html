template<int Dim>
void hpfem::io::write_vtu_facets ( const mesh::Mesh< Dim > &  mesh,
const std::filesystem::path &  file,
bool  boundary_only = true,
VtkFormat  format = VtkFormat::kBinary 
)

Writes the facets of a mesh (boundary only, or all) as lines / triangles with the cell data facet_tag (Int32), facet_index (Int64) and is_boundary (UInt8), to inspect boundary conditions and interfaces. Points are the mesh vertices (and edge nodes for order 2), so the file overlays the cell file exactly.
